cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(xbox_guest_preflight LANGUAGES CXX)
    include(CTest)
endif()
add_executable(xbox_guest_preflight_test preflight_test.cpp)
target_compile_features(xbox_guest_preflight_test PRIVATE cxx_std_20)
target_include_directories(xbox_guest_preflight_test PRIVATE ../../src)
if(MSVC)
    target_compile_options(xbox_guest_preflight_test PRIVATE /W4 /WX /permissive- /EHsc)
else()
    target_compile_options(xbox_guest_preflight_test PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME phase5.guest-preflight COMMAND xbox_guest_preflight_test)
