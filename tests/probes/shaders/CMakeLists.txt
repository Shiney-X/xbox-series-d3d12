cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(xbox_shader_translation LANGUAGES CXX)
    include(CTest)
endif()
include(FetchContent)
set(SPIRV_CROSS_CLI OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_C_API OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_MSL OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_CPP OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_REFLECT OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_UTIL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(spirv_cross
    URL https://codeload.github.com/KhronosGroup/SPIRV-Cross/tar.gz/aa217aeb6c9f0ace7a0ab233b28807edf45eb165
    URL_HASH SHA256=b99db6ee6182dbeb102257e755bf6d4978c23dcfd3329033304779279bc27212)
FetchContent_MakeAvailable(spirv_cross)
add_library(xbox_spirv_hlsl STATIC
    ../../../platform/xbox_uwp/shaders/spirv_hlsl_bridge.cpp
    ../../../platform/xbox_uwp/shaders/compute_fixture.cpp)
target_compile_features(xbox_spirv_hlsl PUBLIC cxx_std_20)
target_include_directories(xbox_spirv_hlsl PUBLIC ../../../platform/xbox_uwp/shaders ../../../src)
target_link_libraries(xbox_spirv_hlsl PRIVATE spirv-cross-hlsl)
if(MSVC)
    target_compile_options(xbox_spirv_hlsl PRIVATE /W4 /WX /permissive- /EHsc)
else()
    target_compile_options(xbox_spirv_hlsl PRIVATE -Wall -Wextra -Werror)
endif()
if(BUILD_TESTING)
    add_executable(xbox_spirv_hlsl_test translation_test.cpp)
    target_link_libraries(xbox_spirv_hlsl_test PRIVATE xbox_spirv_hlsl)
    add_test(NAME phase4.spirv-hlsl COMMAND xbox_spirv_hlsl_test)
    set_tests_properties(phase4.spirv-hlsl PROPERTIES TIMEOUT 30)
endif()
